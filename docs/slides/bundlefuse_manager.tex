\begin{frame}{Problem statement}
\key{Complete a partially observed bundle.}
\vspace{5mm}
\[
 \underbrace{S=\{\text{shirt, trousers}\}}_{\text{observed items}}
 \quad\longrightarrow\quad
 \underbrace{c\in\mathcal I\setminus S}_{\text{candidate completion}}
\]
\vspace{5mm}
\[
 \boxed{\widehat C_K(S)=\operatorname{TopK}_{c\notin S}\;\theta(c\mid S)}
\]
\vspace{4mm}
$\mathcal I$: item catalogue.\quad $K$: number of recommendations.\par
\vspace{3mm}
\textbf{Goal:} rank items that belong with the observed combination.
\end{frame}

\begin{frame}{Problem statement}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.47\textwidth}
\key{Specific historical evidence}\par\vspace{4mm}
These items were selected together.\par\vspace{4mm}
\textbf{Limitation:} some candidates have no recorded matches.\par\vspace{3mm}
\textbf{Example:} a new shoe has no recorded co-occurrence with this shirt or these trousers.
\end{column}
\begin{column}{0.47\textwidth}
\key{Transferable descriptive evidence}\par\vspace{4mm}
These items have related attributes.\par\vspace{4mm}
\textbf{Limitation:} resemblance need not imply compatibility.\par\vspace{3mm}
\textbf{Example:} trousers resembling the observed trousers may rank highly, even when the missing item is a shoe.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Research question}
\key{Can recorded bundle occurrences and item attributes support accurate
completion without gradient-training a scoring model?}
\vspace{6mm}
\begin{enumerate}\setlength\itemsep{8pt}
\item \textbf{Use observed combinations}\par
Distinguish several query items occurring together from separate pairwise matches.
\item \textbf{Handle missing item-pair observations}\par
Score a candidate even when it has never occurred with the query items.
\item \textbf{Test the value of combining both}\par
Compare with each source collection alone and with published neural results.
\end{enumerate}
\end{frame}

\begin{frame}{BundleFuse}
\begin{enumerate}\setlength\itemsep{13pt}
\item \key{Parameter-free scoring core}\par
Zero gradient-trained scoring parameters.
\item \key{No neural-network training}\par
Compute statistics and representations; select scalar settings on validation.
\item \key{Historical bundles as completion examples}\par
Preserve joint evidence, transfer it through shared relations, and
adjust reliance using candidate history.
\end{enumerate}
\end{frame}

\begin{frame}{General scoring equations}
\textbf{Final score:} mix all selected scores with a specified subset.
\vspace{4mm}
\[
 \boxed{\theta(c\mid S)=\frac{f_c}{f_c+\kappa}Y_c+
                      \frac{\kappa}{f_c+\kappa}s_0P_c}
\]
\vspace{3mm}
\begin{tabular}{@{}p{0.10\textwidth}p{0.85\textwidth}@{}}
$Y_c$ & Combination of all selected score sources\\[5pt]
$P_c$ & Category, descriptor or metadata scores, plus selected popularity; the chosen sources depend on the dataset\\[5pt]
$f_c$ & Candidate's number of appearances in training bundles\\[5pt]
$s_0,\kappa$ & Fallback scale and frequency threshold, respectively
\end{tabular}\par
\vspace{4mm}
\key{At $f_c=0$, only the scaled subset score $s_0P_c$ remains.}
\end{frame}

\begin{frame}{General scoring equations}
\textbf{Combine standardized sources $z_h$ with scalar weights.}
\[
 Y_c=\sum_{h\in\mathcal H}w_hz_h(c\mid S),\qquad
 P_c=\sum_{h\in\mathcal P}\widetilde w_hz_h(c\mid S),\quad\mathcal P\subseteq\mathcal H.
\]
$\mathcal H$: available sources.\quad $\mathcal P$: chosen fallback sources.\par
\vspace{6mm}
$z_h(c\mid S)$: standardized score from source $h$.\par\vspace{4mm}
$w_h,\widetilde w_h$: scalar weights selected for the two combinations.\par\vspace{6mm}
\key{The fallback combines selected sources already available to the full score.}
\end{frame}

\begin{frame}{General scoring equations}
\textbf{Put different score sources on a comparable scale.}
\vspace{5mm}
\[
 \boxed{z_h(c\mid S)=\frac{r_h(c\mid S)-\mu_h(S)}{\sigma_h(S)+\epsilon}}
\]
\vspace{5mm}
\begin{tabular}{@{}ll@{}}
$r_h$ & Raw score computed from history and available attributes\\[4pt]
$\mu_h(S)$ & Mean of this score over catalogue items, for query $S$\\[4pt]
$\sigma_h(S)$ & Standard deviation over the same items\\[4pt]
$\epsilon>0$ & Small constant for numerical stability
\end{tabular}\par
\vspace{5mm}
\key{No hidden completion labels enter these normalization statistics.}
\end{frame}

\begin{frame}{Ranking metrics at 20}
Let $G$ be the withheld items and $L_{20}$ the top 20 recommendations.
\[
 \boxed{\mathrm{Recall@20}=\frac{|L_{20}\cap G|}{|G|}}
\]
\textbf{Recall:} fraction of withheld items recovered in the top 20.\par
\vspace{3mm}
Let $\mathrm{rel}_r=1$ if rank $r$ is withheld, and $0$ otherwise.
\[
 \mathrm{DCG@20}=\sum_{r=1}^{20}\frac{\mathrm{rel}_r}{\log_2(r+1)},
 \qquad
 \boxed{\mathrm{NDCG@20}=\frac{\mathrm{DCG@20}}{\mathrm{IDCG@20}}}
\]
\textbf{NDCG:} rewards withheld items appearing earlier in the ranking.\par
\vspace{2mm}
$\mathrm{IDCG@20}$ is the DCG of the ideal ordering. Both metrics are averaged over queries.
\end{frame}
